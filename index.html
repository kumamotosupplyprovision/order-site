function doPost(e) { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('全ログ') || ss.getSheets()[0]; var userList = { "kouda28": { pass: "drc", name: "高田会衆" }, "kagami15": { pass: "drc", name: "鏡会衆" }, "uki31": { pass: "drc", name: "宇城会衆" }, "sample01": { pass: "drc", name: "サンプル会衆" } }; var user = userList[data.userId]; if (!user || user.pass !== data.password) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": "パスワードが違うで！" })).setMimeType(ContentService.MimeType.JSON); } if (data.action === "login") { return ContentService.createTextOutput(JSON.stringify({ "result": "success", "kaishuName": user.name })).setMimeType(ContentService.MimeType.JSON); } if (data.action === "order") { var date = new Date(); var name = data.name; var phone = data.phone || ""; var address = data.address; var items = data.items; var comment = data.comment || ""; var kaishu = data.kaishuName || ""; // ★追加：写真用の配列（H列以降に順番に入れるための箱）★ var imageFormulas = []; if (data.images && data.images.length > 0) { var folderIter = DriveApp.getFoldersByName("レスキュー物資画像"); var targetFolder = folderIter.hasNext() ? folderIter.next() : DriveApp.createFolder("レスキュー物資画像"); for (var j = 0; j < data.images.length; j++) { var imgInfo = data.images[j]; var blob = Utilities.newBlob(Utilities.base64Decode(imgInfo.base64), "image/jpeg", imgInfo.name + ".jpg"); var file = targetFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // ★魔法：スプレッドシートのセルに直接画像を表示する関数を作る！★ var fileId = file.getId(); // 画像を直接シートに呼び出すための特殊なURL var directUrl = "https://drive.google.com/uc?export=view&id=" + fileId; imageFormulas.push('=IMAGE("' + directUrl + '")'); // 依頼した人（スマホ側）の履歴用に、文字のリンクも残しておく items = items.replace(imgInfo.name, imgInfo.name + " \n[写真👀: " + file.getUrl() + "]"); } } // A〜G列までの基本データ var rowData = [date, name, phone, address, items, comment, kaishu]; // その後ろ（H列以降）に、写真の関数をガッチャンコ！ rowData = rowData.concat(imageFormulas); sheet.appendRow(rowData); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } if (data.action === "history") { var allData = sheet.getDataRange().getDisplayValues(); var historyList = []; for (var i = 0; i < allData.length; i++) { var colB = allData[i][1]; var colG = allData[i].length > 6 ? allData[i][6] : ""; if (colB === data.name || colG === data.name) { var itemData = allData[i][4]; if (!itemData || itemData === "") { itemData = allData[i][3]; } historyList.push({ date: allData[i][0], items: itemData }); } } historyList.reverse(); return ContentService.createTextOutput(JSON.stringify({ "result": "success", "history": historyList })).setMimeType(ContentService.MimeType.JSON); } }
